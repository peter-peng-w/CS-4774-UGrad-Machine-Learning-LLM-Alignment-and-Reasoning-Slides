\begin{frame}{The LLM as a policy}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{itemize}\setlength{\itemsep}{5pt}

  \item \textbf{Action $a_t$} = the next token $y_t\in V$

  \item \textbf{Episode} = one complete response $y = (y_1,\dots,y_T)$
  \item \textbf{Reward} $R(x,y)$ = a score for the \emph{whole} response, given once at the end
\end{itemize}
\end{column}

\begin{frame}{Policy gradient: how to increase a reward you can only sample}
We cannot backprop through $R$ (a human judgment or a black-box checker). But we can move probability mass toward responses that scored well:
\[
  \nabla_\theta J(\theta) = \E_{y\sim\pi_\theta(\cdot\mid x)}\Big[\, R(x,y)\; \nabla_\theta \log \pi_\theta(y\mid x) \,\Big],
  \qquad \log\pi_\theta(y\mid x) = \sum_t \log p_\theta(y_t\mid x, y_{<t})
\]
\textbf{REINFORCE} (Williams, 1992): sample a response $y$, then
\[
  \theta \leftarrow \theta + \alpha\, R(x,y)\, \nabla_\theta \log \pi_\theta(y\mid x).
\]
\begin{itemize}\setlength{\itemsep}{4pt}
  \item Good response ($R$ large): make every token of it more likely. Bad response: less likely.
  \item Needs no labels and no differentiable reward --- just samples and scores.
\end{itemize}
\vspace{2pt}
\scriptsize\gray{One-line derivation: $\nabla_\theta \sum_y \pi_\theta(y) R(y) = \sum_y \pi_\theta(y)\,\nabla_\theta\log\pi_\theta(y)\, R(y)$ because $\nabla\pi = \pi\,\nabla\log\pi$.}
\end{frame}

\begin{frame}{Read it as: sample $\to$ score $\to$ weighted SFT step}
\begin{center}
\renewcommand{\arraystretch}{1.6}
\begin{tabular}{@{}lll@{}}
\toprule
 & \textbf{gradient} & \textbf{where $y$ comes from}\\
\midrule
\textbf{SFT} (Part 4) & $1 \cdot \nabla_\theta \log p_\theta(y^\star\mid x)$ & a \emph{human} wrote $y^\star$\\
\textbf{RL} & $\textcolor{uvaorange!85!black}{R(x,y)} \cdot \nabla_\theta \log \pi_\theta(y\mid x)$ & the \emph{model} sampled $y$\\
\bottomrule
\end{tabular}
\end{center}
\vspace{6pt}
\keybox{\textbf{Policy gradient = the SFT gradient on self-generated data, weighted by how good it was.}}
\vspace{6pt}
\begin{itemize}\setlength{\itemsep}{4pt}
  \item Nothing changes in the network, the softmax head, or the optimizer.
  \item What changes: the data is the model's own output, and each sequence carries a \emph{weight}.
\end{itemize}
\end{frame}

\begin{frame}{Baselines and the advantage: good \emph{relative to what}?}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\begin{itemize}\setlength{\itemsep}{5pt}
  \item If every response scores $0.6$--$0.9$, \emph{all} get pushed up; only differences carry information, and one sample is very noisy.
  \item Fix: subtract a \textbf{baseline} $b(x)$ --- unchanged expected gradient, much lower variance:
  \[ \nabla_\theta J = \E\big[(R(x,y) - b(x))\,\nabla_\theta\log\pi_\theta(y\mid x)\big] \]
  \item \textbf{Advantage} $A = R - b$: ``how much better than expected''.
  \item $b$ from a learned value function (``critic'') in classic RL; from a \emph{group of samples} in GRPO (Part 7).
\end{itemize}
\end{column}

\begin{frame}{Guardrail 1: don't move too far in one step (PPO clipping)}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{itemize}\setlength{\itemsep}{6pt}
  \item The policy \emph{generates its own training data}: one step too far $\to$ garbage samples $\to$ worse step. (In supervised learning the next batch corrects a bad step.)
  \item \textbf{PPO} (2017): per token, let $\rho_t = \dfrac{\pi_\theta(y_t\mid\cdot)}{\pi_{\text{old}}(y_t\mid\cdot)}$ and optimize
  \[ \min\big(\rho_t A,\; \mathrm{clip}(\rho_t, 1-\epsilon, 1+\epsilon)\,A\big). \]
  \item Once a token's probability has moved by $\pm\epsilon$ (e.g.\ 20\%), it gets no further push this round.
\end{itemize}
\end{column}

\begin{frame}{Guardrail 2: stay close to where you started (KL penalty)}
\begin{itemize}\setlength{\itemsep}{8pt}
  \item Subtract a penalty for drifting away from a frozen \textbf{reference model} $\pi_{\text{ref}}$ (usually the SFT model):
  \[
    \max_\theta\; \E_{y\sim\pi_\theta}\big[R(x,y)\big] \;-\; \beta\,\mathrm{KL}\big(\pi_\theta(\cdot\mid x)\,\|\,\pi_{\text{ref}}(\cdot\mid x)\big)
  \]
  \item Keeps the language fluent, prevents collapsing onto one answer, and limits exploiting flaws in the reward.
  \item $\beta$ trades off ``get more reward'' against ``stay a normal language model''.
\end{itemize}
\vspace{10pt}
\callback{09/15 regularization}{$L_2$ keeps the weights $w$ near $0$; the KL term keeps the \emph{policy} $\pi_\theta$ near $\pi_{\text{ref}}$. Same idea, different space.}
\end{frame}
